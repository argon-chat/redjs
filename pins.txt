# The native inputs redjs is compiled against, pinned in ONE place.
#
#   setup.ps1        fetches each of them, verifies the hash, and (re)extracts
#                    when what is on disk is not what is pinned here;
#   redjs/build.rs   refuses to build against anything else — a prebuilt or a
#                    mimalloc tree that does not match is a build error, never a
#                    silent link of the wrong engine.
#
# Bumping is: edit this file, run ./setup.ps1, run ./build.ps1 -Test.
# Format: `key = value`, `#` starts a comment.

# What the prebuilt says about itself: `BUN_WEBKIT_VERSION` in

# oven-sh/mimalloc, the allocator this JSC build calls

mimalloc.tree-sha256 = e7dd762d39af37c521042cbb2023a2f0aa02d20635df277e604b14ca27f20ebc
