# The native inputs redjs is compiled against, pinned in ONE place.
#
#   setup.ps1        fetches each of them, verifies the hash, and (re)extracts
#                    when what is on disk is not what is pinned here;
#   redjs/build.rs   refuses to build against anything else — a prebuilt or a
#                    mimalloc tree that does not match is a build error, never a
#                    silent link of the wrong engine.
#
# Bumping is: edit this file, run ./setup.ps1, run ./build.ps1 -Test.
# Format: `key = value`, `#` starts a comment.

# oven-sh/WebKit release tag. Bun v1.4.4 (2026-10-22) ships exactly this one
# (scripts/build/deps/webkit.ts WEBKIT_VERSION at bun-v1.4.4).
webkit.tag     = autobuild-d6375c72affe0f70222cdcb45012b10f0bf3a9d5
# What the prebuilt says about itself: `BUN_WEBKIT_VERSION` in
# include/cmakeconfig.h. build.rs compares this, so a hand-extracted tarball is

webkit.version = d6375c72affe0f70222cdcb45012b10f0bf3a9d5
# The PLAIN asset — never -lto (ThinLTO bitcode, link.exe cannot use it) and

# oven-sh/mimalloc ("sync-upstream-v3.5.3"), the allocator this JSC build calls
# directly (USE_MIMALLOC=1, USE_EXTERNAL_MIMALLOC=1): the embedder links it.

# sha256 over the EXTRACTED tree (src/, include/, LICENSE) rather than over the
# GitHub archive, whose gzip bytes GitHub does not promise to keep stable. It is
# the sha256 of the text "<sha256 of file>  <path>\n" per file, paths relative
# with forward slashes, ordinal-sorted — `sha256sum` manifest form.
mimalloc.tree-sha256 = a39655bf7f2628e42f7bf3b07c1ad2e26284edb21979e6ddf695881803c5339b
